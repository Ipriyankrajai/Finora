     - Always: userId = ctx.session.user.id
     - If datePreset: Calculate date range (e.g., last30days = now - 30 days)
     - If dateFrom/dateTo: Use custom range
     - If type: Filter by type
     - If tagId: Use tags: { some: { tagId: input.tagId } }
     - If amountMin/amountMax: Convert to BigInt and filter amountCents
   - Fetch limit + 1 to determine hasMore
   - Include tags relation for display
   - Order by date desc (most recent first)
   - Return { items, nextCursor }

2. **getById**: Query for single transaction with tags
   - Check userId match, throw UNAUTHORIZED if mismatch

3. **create**: Mutation creating transaction with tags
   - Convert amount string to cents using displayToCents
   - Create transaction with userId from session
   - If tagIds provided: Create TransactionTag entries
   - Use Prisma transaction for atomicity
   - Return created transaction with tags

4. **update**: Mutation updating transaction
   - Find transaction first, check ownership
   - If tagIds provided: Delete existing TransactionTags, create new ones
   - Use Prisma transaction for atomicity
   - Convert amount to cents if provided

5. **delete**: Mutation hard-deleting transaction
   - Find transaction first, check ownership
   - Hard delete (not soft delete - transactions don't have isActive)
   - Cascade will handle TransactionTag cleanup

function getDateRangeFromPreset(preset: string): { from: Date; to: Date } {
  const now = new Date();
  const to = now;
  let from: Date;

  switch (preset) {
    case "last7days":
      from = new Date(now.setDate(now.getDate() - 7));
      break;
    // ... etc
  }
  return { from, to };
}
