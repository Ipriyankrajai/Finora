- Input: { id, type?, amount?, date?, description?, tagIds? }
- Implement optimistic update similar to create
- onSuccess: Show success toast "Transaction updated"

**useDeleteTransaction():**
- Use useMutation with trpc.transaction.delete
- Input: { id }
- Implement optimistic update (remove from cache)
- onSuccess: Show success toast "Transaction deleted"

const transactionSchema = z.object({
  type: z.enum(["INCOME", "EXPENSE"]),
  amount: z.string()
    .min(1, "Amount is required")
    .refine(val => /^\d+(\.\d{1,2})?$/.test(val), "Enter a valid amount")
    .refine(val => parseFloat(val) > 0, "Amount must be positive")
    .refine(val => parseFloat(val) <= 999999999.99, "Amount too large"),
  date: z.date({ required_error: "Date is required" }),
  description: z.string().max(500).optional().transform(v => v || null),
  tagIds: z.array(z.string()).optional(),
});
