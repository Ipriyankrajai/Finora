Per CONTEXT.md: Store filters in URL search params for shareable/bookmarkable filters.

Create hook that manages filter state via URL:

```typescript
interface TransactionFilters {
  datePreset?: "last7days" | "last30days" | "thisMonth" | "lastMonth" | "thisYear";
  dateFrom?: string;  // ISO date string
  dateTo?: string;    // ISO date string
  type?: "INCOME" | "EXPENSE";
  tagId?: string;
  amountMin?: string;
  amountMax?: string;
  page: number;
}

function useTransactionFilters() {
  const searchParams = useSearchParams();
  const router = useRouter();
  const pathname = usePathname();

  // Parse current filters from URL
  const filters: TransactionFilters = {
    datePreset: searchParams.get("datePreset") as TransactionFilters["datePreset"],
    dateFrom: searchParams.get("dateFrom") ?? undefined,
    dateTo: searchParams.get("dateTo") ?? undefined,
    type: searchParams.get("type") as TransactionFilters["type"],
    tagId: searchParams.get("tagId") ?? undefined,
    amountMin: searchParams.get("amountMin") ?? undefined,
    amountMax: searchParams.get("amountMax") ?? undefined,
    page: parseInt(searchParams.get("page") ?? "1", 10),
  };

  // Set individual filter (auto-resets to page 1)
  const setFilter = (key: keyof TransactionFilters, value: string | null) => { ... };

  // Clear all filters
  const clearFilters = () => router.push(pathname);

  // Get active filter count for UI badge
  const activeFilterCount = computed from non-empty filters (excluding page);

  return { filters, setFilter, clearFilters, activeFilterCount };
}
```

Important: When any filter changes (except page), reset page to 1 to avoid empty results.

Create hook that fetches transactions with current filters:

```typescript
function useTransactions() {
  const { filters } = useTransactionFilters();

  // Convert page to cursor-based pagination for API
  // API uses cursor, but we expose page numbers to users per CONTEXT.md
  const limit = 20;

  const query = trpc.transaction.list.useQuery({
    ...filters,
    limit,
    // Note: For page-based pagination on cursor API:
    // Page 1: no cursor
    // Page 2+: Need to track cursors from previous pages
  });

  // Group transactions by date for display
  const groupedTransactions = useMemo(() => {
    return groupTransactionsByDate(query.data?.items ?? []);
  }, [query.data?.items]);

  return {
    groups: groupedTransactions,
    isLoading: query.isLoading,
    error: query.error,
    hasNextPage: query.data?.hasMore ?? false,
    refetch: query.refetch,
  };
}
```

Note: The API uses cursor pagination but CONTEXT.md specifies page numbers for UI.
